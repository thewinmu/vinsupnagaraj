// ระบบสุ่มแท้จริง (Fisher-Yates Crypto Shuffle) function shuffleArray(array) { let m = array.length, t, i; while (m) { // ใช้ crypto.getRandomValues เพื่อความแม่นยำในการสุ่มระดับสูง const randomBuffer = new Uint32Array(1); window.crypto.getRandomValues(randomBuffer); i = Math.floor((randomBuffer[0] / (0xffffffff + 1)) * m--); t = array[m]; array[m] = array[i]; array[i] = t; } return array; } function pickSacredCard(idx) { if (!categoryCards || categoryCards.length === 0) { alert("กำลังประสานสาส์นนำทางจากตาราง กรุณาลองใหม่อีกครั้งครับ"); return; } // 1. สับสำรับไพ่ 100 รายการแบบสุ่มแท้จริง const shuffledDeck = shuffleArray([...categoryCards]); // 2. จับใบที่สุ่มได้ตามที่ผู้ใช้เลือก const card = shuffledDeck[idx % shuffledDeck.length]; // 3. แสดงผลคำทำนาย document.getElementById("res-card-name").innerText = card.name || "สาส์นญาณบารมีนาคราช"; document.getElementById("res-card-cat").innerText = `หมวด: ${card.category || currentCategory}`; document.getElementById("res-card-meaning").innerText = card.meaning || "ตั้งมั่นอยู่ในความดี มีสติในการตัดสินใจ แล้วทุกสิ่งจะราบรื่น"; document.getElementById("res-card-advice").innerText = card.advice || "หมั่นรักษาศีล เจริญภาวนา แผ่เมตตาจิต และมีสติรู้เท่าทันอารมณ์ในปัจจุบัน"; document.getElementById("result-modal").classList.remove("hidden"); }
